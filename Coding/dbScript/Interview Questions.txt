Q0: All records
Ans: SELECT e.*, d.* FROM departments d INNER JOIN employees e ON e.dept_id = d.dept_id;

Q1: How do you find duplicate records like name, salaries, degrees?
Ans: SELECT emp_name, COUNT(*) AS count FROM employees GROUP BY emp_name HAVING COUNT(*) > 1;
     SELECT degree, COUNT(*) AS count FROM employees GROUP BY degree HAVING COUNT(*) > 1;

Q2: How do you find the second highest salary?
Ans: SELECT DISTINCT salary AS SecondHighestSalary FROM employees ORDER BY salary DESC LIMIT 1 OFFSET 1;

Q3: Find employees who earn more than their manager.
Ans: SELECT e.emp_name, e.salary, e.city AS employee_city, m.emp_name AS Manager, m.salary, d.dept_name, d.location AS Dpt_Location FROM employees e INNER JOIN employees m ON e.manager_id = m.emp_id INNER JOIN departments d ON e.dept_id = d.dept_id where e.salary > m.salary;

Q4: Find the number of employees in each department
Ans: SELECT d.dept_id, d.dept_name, COUNT(e.emp_id) AS employee_count FROM departments d INNER JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_id, d.dept_name;

Q5: Find departments having more than 2 employees
Ans: SELECT d.dept_id, d.dept_name, COUNT(e.emp_id) AS employee_count FROM departments d INNER JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_id, d.dept_name HAVING COUNT(e.emp_id) > 2;

Q6: Find the highest-paid employee in each department
Ans: SELECT d.dept_name, e.emp_name, e.salary FROM employees e JOIN departments d ON e.dept_id = d.dept_id WHERE e.salary = (SELECT MAX(e2.salary) FROM employees e2 WHERE e2.dept_id = e.dept_id);

Q7: Find employees earning more than the average salary of their department
Ans: SELECT e.emp_id, e.emp_name, e.salary, d.dept_name FROM employees e JOIN departments d ON e.dept_id = d.dept_id WHERE e.salary > (SELECT AVG(e2.salary) FROM employees e2 WHERE e2.dept_id = e.dept_id);

Q8: Find employees along with their manager name 
Ans: SELECT e.emp_id, e.emp_name, e.city AS employee_city, m.emp_name AS Manager, d.dept_name, d.location AS Dpt_Location FROM employees e INNER JOIN employees m ON e.manager_id = m.emp_id INNER JOIN departments d ON e.dept_id = d.dept_id;

Q9: Find the employees having greater then avg employees salary
Ans: SELECT * FROM employees WHERE salary > (SELECT AVG(salary) FROM employees);

Q10: Find departments max salary
Ans: SELECT dept_id, MAX(salary) AS max_salary FROM employees GROUP BY dept_id;
